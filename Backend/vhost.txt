    keyFile                 /etc/letsencrypt/live/adda.skillinnovex.in/privkey.pem

}

    location                /usr/local/lsws/Example/html/.well-known/acme-challenge

    type                    proxy
    address                 127.0.0.1:3015
    maxConns                100
    pcKeepAliveTimeout      60
    initTimeout             60
    retryTimeout            0
    respBuffer              0
}

    enable                  1
    autoLoadHtaccess        0

    # Canonical protocol: http -> https
    RewriteCond %{HTTPS} !=on
    RewriteRule ^(.*)$ https://adda.skillinnovex.in/$1 [R=301,L]

    # Proxy everything (API, /media, /health, /ready) to the AddaApp Node backend
    RewriteRule ^(.*)$ http://addaapi/$1 [P,L]
